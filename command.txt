for model in pfnet gns canos; do
  echo "===== ${model} case118 finished runs ====="

  find runs -path "*${model}*" -name summary.json | while read s; do
    d="$(dirname "$s")"
    if grep -q "case_name: case118" "$d/config.yaml" 2>/dev/null \
      && grep -q "TRAINING FINISHED" "$d/out.txt" 2>/dev/null; then
      task="$(grep -m1 "task:" "$d/config.yaml" | awk '{print $2}')"
      seed="$(grep -m1 "seed:" "$d/config.yaml" | awk '{print $2}')"
      echo "task=${task} seed=${seed} dir=${d}"
    fi
  done | sort

  echo
done
